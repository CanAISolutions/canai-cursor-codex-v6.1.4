    // Check Redis cache first
    const cached = await redis.get(cacheKey);

    target_educational_impact DECIMAL DEFAULT 0.90,
    target_canai_selection DECIMAL DEFAULT 0.85
) RETURNS TABLE (

    current_value DECIMAL,
    target_value DECIMAL,
