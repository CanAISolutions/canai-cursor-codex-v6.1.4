export class BusinessPlanEmotionalValidator {

  /**
   * Validate business plan empathy across 4 dimensions
   */
  async validateBusinessPlanEmpathy(content: string, input: any): Promise<EmpathyValidationResult> {
    const emotionalResonance = await this.calculateEmotionalResonance(content, input);
    const toneAlignment = await this.calculateToneAlignment(content, input);
    const connectionStrength = await this.calculateConnectionStrength(content, input);
    const authenticity = await this.calculateAuthenticity(content, input);

    const overallScore = (emotionalResonance + toneAlignment + connectionStrength + authenticity) / 4;

    return {
      emotionalResonance,
      toneAlignment,
      connectionStrength,
      authenticity,
      overallScore,
      passesThreshold: overallScore >= 0.85
    };
  }

  /**
   * Calculate emotional resonance based on content and context
   */
  private async calculateEmotionalResonance(content: string, input: any): Promise<number> {
    let score = 0.7; // Base score

    // Check for emotional language
    const emotionalWords = ['passion', 'vision', 'dream', 'impact', 'transform', 'empower', 'inspire'];
    const emotionalCount = emotionalWords.filter(word => content.toLowerCase().includes(word)).length;
    score += (emotionalCount * 0.02);

    // Check for personal story integration
    if (input.emotionalContext?.personalStory && content.includes('story')) score += 0.1;
    if (input.emotionalContext?.visionQuote && content.includes('vision')) score += 0.1;

    // Check for audience empathy
    if (input.targetAudience && content.toLowerCase().includes('understand')) score += 0.05;
    if (content.includes('pain point') || content.includes('challenge')) score += 0.05;

    return Math.min(score, 1.0);
  }

  /**
   * Calculate tone alignment with brand voice
   */
  private async calculateToneAlignment(content: string, input: any): Promise<number> {
    const brandVoice = input.brandVoice || input.tone || 'professional';
    let score = 0.8; // Base alignment score

    switch (brandVoice.toLowerCase()) {
      case 'professional':
        if (content.includes('strategic') || content.includes('analysis')) score += 0.1;
        if (content.includes('slang') || content.includes('casual')) score -= 0.2;
        break;
      case 'enthusiastic':
        if (content.includes('exciting') || content.includes('opportunity')) score += 0.1;
        if (content.includes('conservative') || content.includes('traditional')) score -= 0.1;
        break;
      case 'supportive':
        if (content.includes('guidance') || content.includes('support')) score += 0.1;
        if (content.includes('aggressive') || content.includes('competitive')) score -= 0.1;
        break;
      case 'authoritative':
        if (content.includes('expertise') || content.includes('proven')) score += 0.1;
        if (content.includes('uncertain') || content.includes('maybe')) score -= 0.2;
        break;
    }

    return Math.min(score, 1.0);
  }

  /**
   * Calculate connection strength with target audience
   */
  private async calculateConnectionStrength(content: string, input: any): Promise<number> {
    let score = 0.7;

    // Check for audience-specific language
    const targetAudience = (input.targetAudience || '').toLowerCase();
    if (targetAudience.includes('small business') && content.includes('small business')) score += 0.1;
    if (targetAudience.includes('enterprise') && content.includes('enterprise')) score += 0.1;
    if (targetAudience.includes('startup') && content.includes('startup')) score += 0.1;

    // Check for industry-specific terminology
    const industry = (input.competitiveContext || input.industry || '').toLowerCase();
    if (industry.includes('tech') && content.includes('technology')) score += 0.05;
    if (industry.includes('healthcare') && content.includes('patient')) score += 0.05;
    if (industry.includes('finance') && content.includes('financial')) score += 0.05;

    // Check for problem-solution alignment
    if (input.businessDescription && content.includes('solution')) score += 0.1;

    return Math.min(score, 1.0);
  }

  /**
   * Calculate authenticity based on personal context integration
   */
  private async calculateAuthenticity(content: string, input: any): Promise<number> {
    let score = 0.8;

    // Check for personal context integration
    if (input.emotionalContext?.personalStory) {
      if (content.includes('experience') || content.includes('journey')) score += 0.1;
    }

    if (input.emotionalContext?.motivator) {
      if (content.includes('motivated') || content.includes('driven')) score += 0.05;
    }

    // Check for specific business details
    if (input.businessName && content.includes(input.businessName)) score += 0.05;
    if (input.currentStatus && content.includes('current')) score += 0.05;

    // Penalize generic language
    if (content.includes('Lorem ipsum') || content.includes('[INSERT]')) score -= 0.3;
    if (content.includes('generic') || content.includes('template')) score -= 0.2;

    return Math.min(score, 1.0);
  }

  /**
   * Implement 5-axis emotional compass with scores 4.0-5.0
   */
  calculateEmotionalCompass(content: string, input: any): EmotionalCompass {
    return {
      clarity: this.scoreClarity(content, input),
      empowerment: this.scoreEmpowerment(content, input),
      trust: this.scoreTrust(content, input),
      joy: this.scoreJoy(content, input),
      alignment: this.scoreAlignment(content, input)
    };
  }

  private scoreClarity(content: string, input: any): number {
    let score = 4.0;
    if (content.length > 500 && content.includes('Executive Summary')) score += 0.3;
    if (content.includes('specific') || content.includes('clear')) score += 0.2;
    if (content.split('\n').length > 10) score += 0.3; // Well-structured
    if (content.includes(input.businessName || '')) score += 0.2;
    return Math.min(score, 5.0);
  }

  private scoreEmpowerment(content: string, input: any): number {
    let score = 4.0;
    if (content.includes('achieve') || content.includes('success')) score += 0.2;
    if (content.includes('opportunity') || content.includes('growth')) score += 0.3;
    if (content.includes('competitive advantage')) score += 0.3;
    if (content.includes('actionable') || content.includes('next steps')) score += 0.2;
    return Math.min(score, 5.0);
  }

  private scoreTrust(content: string, input: any): number {
    let score = 4.2; // Minimum per V4
    if (content.includes('realistic') || content.includes('achievable')) score += 0.2;
    if (content.includes('market research') || content.includes('analysis')) score += 0.2;
    if (input.financials && content.includes('revenue')) score += 0.2;
    if (content.includes('proven') || content.includes('validated')) score += 0.2;
    return Math.min(score, 5.0);
  }

  private scoreJoy(content: string, input: any): number {
    let score = 4.0;
    if (content.includes('exciting') || content.includes('innovative')) score += 0.3;
    if (content.includes('passion') || content.includes('vision')) score += 0.3;
    if (input.emotionalContext?.visionQuote && content.includes('vision')) score += 0.2;
    if (content.includes('transform') || content.includes('impact')) score += 0.2;
    return Math.min(score, 5.0);
  }

  private scoreAlignment(content: string, input: any): number {
    let score = 4.0;
    const goal = (input.primaryGoal || input.goal || '').toLowerCase();
    if (goal && content.toLowerCase().includes(goal)) score += 0.3;
    if (input.planPurpose && content.includes(input.planPurpose)) score += 0.3;
    if (input.targetAudience && content.includes('target')) score += 0.2;
    if (input.brandVoice === 'professional' && content.includes('professional')) score += 0.2;
    return Math.min(score, 5.0);
  }
}

interface EmpathyValidationResult {
  emotionalResonance: number;
  toneAlignment: number;
  connectionStrength: number;
  authenticity: number;
  overallScore: number;
  passesThreshold: boolean;
}

interface EmotionalCompass {
  clarity: number;
  empowerment: number;
  trust: number;
  joy: number;
  alignment: number;
}
