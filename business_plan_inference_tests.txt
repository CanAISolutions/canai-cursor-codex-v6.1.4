// Enhanced MCP Field Inference for Business Plan V4
export function applyMCPEnhancersV4(input: BusinessPlanInputV4): BusinessPlanInputV4 {
  const enhanced = { ...input };

  // Infer businessName from primaryGoal if missing
  if (!enhanced.businessName && enhanced.primaryGoal) {
    const businessNameMatch = enhanced.primaryGoal.match(/for\s+([A-Z][A-Za-z0-9\s&]+)/i);
    enhanced.businessName = businessNameMatch ? businessNameMatch[1].trim() : 'New Venture';
  }

  // Infer brandVoice from industry/competitiveContext
  if (!enhanced.brandVoice && enhanced.competitiveContext) {
    enhanced.brandVoice = inferBrandVoiceFromContext(enhanced.competitiveContext);
  }

  // Infer targetAudience from businessDescription
  if (!enhanced.targetAudience && enhanced.businessDescription) {
    enhanced.targetAudience = inferAudienceFromDescription(enhanced.businessDescription);
  }

  // Auto-enhance Universal Emotional Fields
  if (!enhanced.emotionalResonance) {
    enhanced.emotionalResonance = calculateEmotionalResonance(enhanced);
  }
  
  if (!enhanced.trustScore) {
    enhanced.trustScore = calculateBaseTrustScore(enhanced);
  }
  
  if (!enhanced.qualityIndicators || enhanced.qualityIndicators.length === 0) {
    enhanced.qualityIndicators = generateQualityIndicators(enhanced);
  }
  
  if (!enhanced.viralPotential) {
    enhanced.viralPotential = assessViralPotential(enhanced);
  }

  return enhanced;
}

function inferBrandVoiceFromContext(context: string): string {
  const contextLower = context.toLowerCase();
  if (contextLower.includes('finance') || contextLower.includes('banking')) return 'professional';
  if (contextLower.includes('tech') || contextLower.includes('startup')) return 'enthusiastic';
  if (contextLower.includes('healthcare') || contextLower.includes('medical')) return 'authoritative';
  if (contextLower.includes('education') || contextLower.includes('training')) return 'supportive';
  if (contextLower.includes('creative') || contextLower.includes('design')) return 'conversational';
  return 'professional';
}

function inferAudienceFromDescription(description: string): string {
  const descLower = description.toLowerCase();
  if (descLower.includes('small business')) return 'Small business owners with 5-50 employees';
  if (descLower.includes('enterprise')) return 'Enterprise decision makers and C-suite executives';
  if (descLower.includes('consumer')) return 'General consumers aged 25-45 with disposable income';
  if (descLower.includes('b2b')) return 'Business professionals and procurement teams';
  if (descLower.includes('saas')) return 'Technology-forward businesses seeking digital solutions';
  return 'Target market to be defined based on business model';
}
