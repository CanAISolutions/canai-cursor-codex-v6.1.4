**Mitigation Strategy**:
```javascript
// JSON Flattening Preprocessor (Production-Ready)
function flattenNestedObject(obj, prefix = '', maxDepth = 3) {
  const flattened = {};
  
  for (const [key, value] of Object.entries(obj)) {
    const newKey = prefix ? `${prefix}_${key}` : key;
    
    if (typeof value === 'object' && value !== null && maxDepth > 0) {
      Object.assign(flattened, flattenNestedObject(value, newKey, maxDepth - 1));
    } else {
      flattened[newKey] = value;
    }
  }
  
  return flattened;
}

// Usage in webhook payload:

```
